from django.template.loader import render_to_string

def monthly_challenges_dict_view(request, month):
    try:
        challenges_text = monthly_challenges_dict[month]
        return render(request, "challenges/challenge.html",{
            "text" : challenges_text,
            "month_name" : month.capitalize()
        })
    except:
        response_data = render_to_string("404.html")
        return HttpResponseNotFound(response_data)

from django.http import Http404

def monthly_challenges_dict_view(request, month):
    try:
        challenges_text = monthly_challenges_dict[month]
        return render(request, "challenges/challenge.html",{
            "text" : challenges_text,
            "month_name" : month.capitalize()
        })
    except:
        raise Http404   # this is what we add so the djaon will search for that file and we must put the same name 404.html
