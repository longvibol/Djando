    {% endblock %}

</html>

2- extend the html to the template 

first we need to put the header : {% extends "base.html" %}

then we can inject the from the template: 

------------------------------------------------

{% extends "base.html" %}

All Challenges

    <ul>
      {% for month in months %}
        {% comment %} <li><a href="/challenges/{{month}}">{{ month | title}}</a></li> {% endcomment %}
        <li><a href="{% url "month-challenge" month %}">{{ month | title}}</a></li>
      {% endfor %}
    </ul>

------------------------------------------------

Then we can update to : 

-------------------------- challenges.html 

{% extends "base.html" %}

{%block month_name%}

 Monthly Challenges

{%endblock%}

{% block content %}
    <h1>{{ month_name | title }} Challenges</h1>
    {% if text is not None %}
        <h2>{{text | title}}</h2>
    {% else %}
        <p>There is no challenge for this month yet!</p>
    {% endif %}

{%endblock%}
