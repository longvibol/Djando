// function get key on 
def get_date(post):
    return post['date']

def starting_page(request):
    sorted_posts = sorted(all_posts, key=get_date)
    lastest_posts = sorted_posts[-3:]
    return render(request, "blog/index.html",{
        "posts": lastest_posts
    })
